# 数据结构与算法

## 1 什么是数据结构

  

数据+结构

**什么是数据？**

"一切都不是数据，而一切又都在成为数据"

**什么是结构**

不同的需求对应不同的结构，将数据按照一定的结构去展示出来，数据还是那些数据

---

在一些特定的、特殊的需求场景下，我们以往所学的数据类型，无法基于需求合理地组织数据，此时就需要我们自己设计一套新的数据组织形式来解决问题，这就是**数据结构**。

**数据结构**是一种存储、组织数据的方式，旨在便于访问和修改。没有一种单一的数据结构对所有用途均有效，所以需要知道几种数据结构优势和局限。

#### 3.4.4 指针与数组

在C语言中，指针与数组的关系十分密切 通过数组下标能完城的操作都可以通过指针完成

一般来说，用指针编写的程序比用数组下标编写的程序执行速度快

```c
#include <stdio.h>
int main(){
    int a[]={15,22,67,43,87};
    int *p;
    p=a;
    for(int i=0;i<sizeof(a)/sizeof(a[0]);i++){
        printf("%d\n",a[i]);
    }
    for(int i=0;i<sizeof(a)/sizeof(a[0]);i++){
        printf("%d\n",8(p+i));
    }
    return 0;
}
```

### 3.6 结构体

结构体是一个或多个变量的集合，这些变量可以是不同的类型

`struct 结构体名`

`{`

`数据类型 变量名；`

`数据类型 变量名；`

`};`

```c
#include <stdio.h>
struct point{
    int x;
    int y;
};
typedef struct point point;
int main(){
    point a;
    a.x=1;
    a.y=3;

    printf("a(%d,%d)",a.x,a.y);

    return 0;
}    
```

  

结构体只是一个变量类型，也就是说它可以运用到数组、函数里面，和指针这些互相结合起来，灵活多变

以上的综合运用可以参考（建议放到其他编译器里面进行试验）

```c
#include <stdio.h>
#include <stdlib.h>
//声明一个新的结构体
typedef struct{
	double *e;
}Musy;
//创建三元组，number为三元组的个数
Musy *create(int number){
	Musy *p;
	p=(Musy *)calloc(number,sizeof(Musy));
	if(p==NULL){
		printf("内存分配失败\n");
		exit(0);
	}
	
	for(int i=0;i<number;i++){
		(p+i)->e=(double *)calloc(3,sizeof(double));
		
		if((p+i)->e==NULL){
		printf("内存分配失败\n");
		exit(0);
		}
	}
	
	return p;
}
//初始化三元组，p为三元组的指针，e1,e2,e3为三元组的三个值
int initialize(Musy *p,double e1,double e2,double e3){
	p->e[0]=e1;
	p->e[1]=e2;
	p->e[2]=e3;
	
	return 1;
}
//获取三元组的值l1为三元组的序号，l2为三元组元素的序号
double getelement(Musy *p,int l1,int l2){
	double result;
	
	result=(p+l1-1)->e[l2-1];
	
	return result;
}
//置换三元组的值，l1为三元组的序号，l2为三元组元素的序号，newelement为新的值
void changeelement(Musy *p,int l1,int l2,double newelement){
	(p+l1-1)->e[l2-1]=newelement;
}
//获取某三元组最大值，l1为三元组的序号
double getmax(Musy *p,int l1){
	double max=(p+l1-1)->e[0];
	
	if(max<(p+l1-1)->e[1])
	max=(p+l1-1)->e[1];
	if(max<(p+l1-1)->e[2])
	max=(p+l1-1)->e[2];
	
	return max;
}
//获取某三元组最小值，l1为三元组的序号
double getmin(Musy *p,int l1){
	double min=(p+l1-1)->e[0];
	
	if(min>(p+l1-1)->e[1])
	min=(p+l1-1)->e[1];
	if(min>(p+l1-1)->e[2])
	min=(p+l1-1)->e[2];
	
	return min;
}
//对两个三元组对应分量进行运算
void calculate(Musy *p,char op,Musy *q){
	switch(op){
		case '+':{
			double z1=p->e[0]+q->e[0];
			double z2=p->e[1]+q->e[1];
			double z3=p->e[2]+q->e[2];
			
			printf("对应元素计算结果为：\n");
			printf("%lf %lf %lf",z1,z2,z3);
			
			break;
		}
		case '-':{
			double z1=p->e[0]-q->e[0];
			double z2=p->e[1]-q->e[1];
			double z3=p->e[2]-q->e[2];
			
			printf("对应元素计算结果为：\n");
			printf("%lf %lf %lf",z1,z2,z3);
			break;
		}
		case '*':{
			double z1=p->e[0]*q->e[0];
			double z2=p->e[1]*q->e[1];
			double z3=p->e[2]*q->e[2];
			
			printf("对应元素计算结果为：\n");
			printf("%lf %lf %lf",z1,z2,z3);
			break;
		}
		case '/':{
            double z1 = p->e[0] / q->e[0];
            double z3 = p->e[1] / q->e[1];
            double z2 = p->e[2] / q->e[2];
            break;
        }
	}
}
//显示三元组
void show(Musy *p,int l1){
	printf("\n第一个：e1:%6f  e2:%6f  e3:%6f\n",(p+l1-1)->e[0],(p+l1-1)->e[1],(p+l1-1)->e[2]);
}
//给三元组的各分量同乘一个比例因子
void biliyinzi(Musy *p,int l1,double yinshu){
	(p+l1-1)->e[0]*=yinshu;
	(p+l1-1)->e[1]*=yinshu;
	(p+l1-1)->e[2]*=yinshu;
	
	show((p+l1-1),l1);
}
//销毁三元组
void fire(Musy *p,int number){
	// 释放每个三元组的 e
    for(int t=0; t<number; t++){
    free(p[t].e); // 释放 e 的内存
    }
    free(p); // 释放整个 Musy 数组
    printf("\n当前进程已结束\n");
    exit(0);
                
    
}
```

### 6.1 线性表的定义与特点

由n(n>=0) 个**数据特性相同**的元素构成的有限序列，称为**线性表**

线性表中元素的个数n(n>=0)定义为线性表的长度，当n=0时称之为空表

对于非空的线性表或线性结构，其**特点**是：

**存在唯一的一个被称作“第一个”的数据元素；**

**存在唯一的一个被称作“最后一个”的数据元素；**

**除第一个元素外，结构中的每个数据元素均只有一个前驱；**

**除最后一个元素外，结构中的每个数据元素均只有一个后继。**

### 6.3 线性表的链式表示与实现

线性表链式存储结构的特点是：用一组任意的存储单元存储线性表的数据元素（这组存储单元可以是连续的，也可以是不连续的）；

为了表示每个数据元素ai与其直接后继数据元素ai+1之间的逻辑关系，对数据元素ai来说，除了本身的信息之外，还需要存储一个指示其直接后继的信息（直接后继的存储位置）。这两部分信息组成数据元素ai的存储影响，称为节点（node）；

节点包括两个域：其中存储数据元素信息的称为**数据域**；存储直接后继存储位置的域称为**指针域**；

指针与中存储的信息称作指针或链；

n个节点连接成一个链表即为线性表

#### 6.3.3 单向循环链表

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct chain {
    int data;
    struct chain *next; // 修正链表节点类型定义
} chain;

// 创建链表
chain *create(int elem_number) {
    chain *head = (chain *)malloc(sizeof(chain));
    if (head == NULL) {
        printf("分配内存失败！！！\n");
        exit(0);
    }
    head->data = 0; // 头节点不存储数据，仅作为占位符
    head->next = NULL;

    chain *tail = head; // 尾指针用于构建链表

    for (int i = 0; i < elem_number; i++) {
        chain *new_elem = (chain *)malloc(sizeof(chain));
        if (new_elem == NULL) {
            printf("分配内存失败！！！\n");
            exit(0);
        }

        int e;
        scanf("%d", &e);
        new_elem->data = e;
        new_elem->next = NULL;

        tail->next = new_elem; // 将新节点连接到链表尾部
        tail = new_elem;       // 更新尾指针
    }

    return head;
}

// 找到中间节点并返回后半段链表的起始点
chain *find_mid(chain *head, int elem_number) {
    chain *fast = head->next; // 快指针从第一个有效节点开始
    chain *slow = head->next; // 慢指针也从第一个有效节点开始

    // 快指针每次走两步，慢指针每次走一步
    while (fast != NULL && fast->next != NULL) {
        fast = fast->next->next;
        slow = slow->next;
    }

    // 如果元素个数为偶数，slow指向中间两个节点的第二个；奇数时指向中间节点
    // 返回后半段的起始点
    if (elem_number % 2 == 0) {
        return slow->next; // 偶数时后半段从slow的下一个开始
    } else {
        return slow; // 奇数时后半段从slow开始（包含中间节点）
    }
}

// 反转链表
chain *reverse(chain *head) {
    chain *prev = NULL;
    chain *current = head;
    chain *next_node = NULL;

    while (current != NULL) {
        next_node = current->next; // 保存下一个节点
        current->next = prev;      // 反转当前节点的指针
        prev = current;            // 移动prev到当前节点
        current = next_node;       // 移动current到下一个节点
    }

    return prev; // prev最终指向反转后链表的头节点
}

// 将反转后的后半段链表插入到原链表中
void insert(chain *head, chain *reversed_half) {
    chain *original = head->next; // 指向原链表的第一个有效节点
    chain *insert_node = reversed_half; // 指向反转后链表的头节点

    while (original != NULL && insert_node != NULL) {
        // 保存原链表和反转链表的下一个节点
        chain *original_next = original->next;
        chain *insert_next = insert_node->next;

        // 将反转链表的当前节点插入到原链表的当前节点之后
        original->next = insert_node;
        insert_node->next = original_next;

        // 移动指针到下一个插入位置
        original = original_next;
        insert_node = insert_next;
    }
}

// 打印链表
void print_list(chain *head) {
    head = head->next; // 跳过头节点
    while (head != NULL) {
        printf("%d ", head->data);
        head = head->next;
    }
    printf("\n");
}

// 释放链表内存
void free_list(chain *head) {
    chain *temp;
    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
}

int main() {
    int elemNumber;
    printf("输入链表个数：\n");
    scanf("%d", &elemNumber);

    printf("输入数据（用空格隔开）：");
    chain *T = create(elemNumber);

    // 找到中间节点并获取后半段链表
    chain *half = find_mid(T, elemNumber);

    // 反转后半段链表
    half = reverse(half);

    // 将反转后的后半段链表插入到原链表中
    insert(T, half);

    // 打印最终结果
    printf("最终链表：");
    print_list(T);

    // 释放链表内存
    free_list(T);

    return 0;
}
```

### 7.1 栈的定义与实现

#### 7.1.1 定义

**栈**（stack）是限定仅在表尾进行插入或删除操作的线性表。因此，对栈来说，表尾端有其特殊含义，称为**栈顶**（top），相应地，表头端称为**栈底**（bottom）。不含元素的空表称为空栈。

### 7.2 队列的定义与实现

#### 7.2.1 定义

**队列**（queue）是一种先进先出的线性表。它至允许在表的一端进行插入，在另一头删除元素。在队列中允许插入的一端称为**队尾**（rear），允许删除的一段称为**队头**（front）。

就和排队结账一样，你先进入这个队伍，那么你就先出去。别人只能排在你的后边，后进后出。你结完账了，又进超市买点东西，你又要去结账，你就从末尾再进入这个队伍

#### 7.2.2 顺序结构实现

在队列的顺序结构里面有一个比较重要的，就是**调整队列**

你前面的人结完帐了你肯定要往前走啊，对不对，你要不不动的话，那就会造成假满，前面有一块儿地没利用，对吧。而且放到实际应用里面，你不往前走，人家怎么给你结账啊

```c
int queueFull(Queue *Q){

        int step = 	Q->front;//计算出一个元素要往前挪几步
        for(int i=Q->front;i<=Q->rear;++i){
            Q->data[i-step]=Q->data[i];
        }
        Q->front=0;
        Q->rear=Q->rear-step;
        return 1;
    }

        printf("真的满了\n");
        return 0;
    }
}
```

**对 I 解释**

采用非递归方式重写递归程序时不一定要用栈，但栈是最常用的工具之一，因为它可以方便地模拟递归调用过程中的函数调用栈。以下是几种可能的替代方法：

### 1.使用栈（Stack）

栈是模拟递归最直接的方式，因为递归本质上是利用栈的后进先出（LIFO）特性来保存和恢复函数调用时的状态。在非递归实现中，可以手动维护一个栈来保存函数参数和局部变量的状态。

**示例：递归转非递归（使用栈）**


```c
#include <iostream>
#include <stack>
using namespace std;

void recursiveFunction(int n) {
    if (n > 0) {
        cout << n << endl;
        recursiveFunction(n - 1);
    }
}

void iterativeFunction(int n) {
    stack<int> stk;
    stk.push(n);
    while (!stk.empty()) {
        int current = stk.top();
        stk.pop();
        if (current > 0) {
            cout << current << endl;
            stk.push(current - 1); // 模拟递归调用
        }
    }
}

int main() {
    recursiveFunction(5);
    iterativeFunction(5);
    return 0;
}
```

### 2.使用队列（Queue）

在某些情况下，特别是当递归的顺序可以被改变为广度优先（BFS）时，可以使用队列来代替递归。队列的先进先出（FIFO）特性适合处理层次遍历等问题。

**示例：递归转非递归（使用队列）**

```c
#include <iostream>
#include <queue>
using namespace std;

void recursiveFunction(int n) {
    if (n > 0) {
        cout << n << endl;
        recursiveFunction(n - 1);
    }
}

void iterativeFunction(int n) {
    queue<int> q;
    q.push(n);
    while (!q.empty()) {
        int current = q.front();
        q.pop();
        if (current > 0) {
            cout << current << endl;
            q.push(current - 1); // 模拟递归调用
        }
    }
}

int main() {
    recursiveFunction(5);
    iterativeFunction(5);
    return 0;
}
```

### 3.使用迭代（Iteration）

对于一些简单的递归问题，可以直接通过循环来实现非递归版本，而不需要显式地使用栈或队列。

**示例：递归转非递归（使用循环）**


```c
#include <iostream>
using namespace std;

void recursiveFunction(int n) {
    if (n > 0) {
        cout << n << endl;
        recursiveFunction(n - 1);
    }
}

void iterativeFunction(int n) {
    while (n > 0) {
        cout << n << endl;
        n--; // 模拟递归调用
    }
}

int main() {
    recursiveFunction(5);
    iterativeFunction(5);
    return 0;
}
```

### 4.使用尾递归优化（Tail Recursion Optimization）

如果递归函数是尾递归（即递归调用是函数的最后一个操作），可以通过尾递归优化将其转换为迭代，而不需要显式地使用栈。

**示例：尾递归转非递归**


```c
#include <iostream>
using namespace std;

void recursiveFunction(int n, int result) {
    if (n == 0) {
        cout << result << endl;
    } else {
        recursiveFunction(n - 1, result * n); // 尾递归
    }
}

void iterativeFunction(int n) {
    int result = 1;
    while (n > 0) {
        result *= n;
        n--;
    }
    cout << result << endl;
}

int main() {
    recursiveFunction(5, 1);
    iterativeFunction(5);
    return 0;
}
```

### 总结

- **栈**：最适合模拟递归调用栈，适用于大多数递归问题。
- **队列**：适用于可以改变递归顺序的问题，特别是广度优先搜索（BFS）。
- **循环**：适用于简单的递归问题，可以直接通过循环实现。
- **尾递归优化**：适用于尾递归函数，可以通过循环实现。

### 11.1 定义

---

#### 11.1.1 树

**树**，是一个或多个节点的**有限集合**

存在一个称为根的特定结点，其余的节点被分为n个互不相交的集合T1，T2。。。Tn，其中每个集合都是一棵树。T1，T2,....Tn称为根节点的子树。

![](https://cdn.nlark.com/yuque/0/2025/png/53984905/1742101674167-beca149d-e7d9-431b-9ce1-8d2fa1d1fdf0.png)

  

**结点：书中的一个独立单元**

**结点的度：节点拥有的子树个数称为结点的度**

**树的度：树内各结点度的最大值**

**叶子：度为0的结点或者终端结点**

**双亲和孩子：节点的子树的根称为该结点的孩子，相应的，该结点称为孩子的双亲**

**层次：节点额层次从根开始定义，根为第一层，跟的孩子为第二层，以此类推**

---

**树的基本性质**

**性质一：**树中所有结点数等于所有结点的度数之和加一

**性质二：**对于度为m的树，第 i 层上最多有 m的（i-1）次个结点

**性质三：**对于高度为h，度为m的树，最多有（m的h次方-1）/（m-1）个结点

---

#### **11.1.2 二叉树**

是n（n>=0)个结点所构成的集合，它或为空树，或为非空树。对于非空树T：

（1）有且仅有一个称为根的结点

（2）除根节点意外的其余节点分为两个互不相交的子集T1和T2，分别成为T的左子树和右子树，且T1和T2本身又都是二叉树

（3）二叉树每个结点之多只有两棵子树

（4）二叉树的子树有左右之分，其次序不能任意颠倒

```c
#include <stdio.h>
#include <stdlib.h>

// 定义二叉树节点结构
typedef struct TreeNode {
    char data;
    struct TreeNode *left;
    struct TreeNode *right;
} TreeNode;

// 创建二叉树（按层输入，空节点用 '#' 表示）
TreeNode* createTree() {
    char ch;
    scanf(" %c", &ch); // 读取一个字符，注意前面的空格用于跳过可能的换行符或空格
    if (ch == '#') {
        return NULL;
    } else {
        TreeNode *node = (TreeNode*)malloc(sizeof(TreeNode));
        node->data = ch;
        printf("请输入 %c 的左子节点: ", ch);
        node->left = createTree();
        printf("请输入 %c 的右子节点: ", ch);
        node->right = createTree();
        return node;
    }
}

// 前序遍历（递归）
void preOrder(TreeNode *root) {
    if (root == NULL) {
        return;
    }
    printf("%c ", root->data);
    preOrder(root->left);
    preOrder(root->right);
}

// 中序遍历（递归）
void inOrder(TreeNode *root) {
    if (root == NULL) {
        return;
    }
    inOrder(root->left);
    printf("%c ", root->data);
    inOrder(root->right);
}

// 后序遍历（递归）
void postOrder(TreeNode *root) {
    if (root == NULL) {
        return;
    }
    postOrder(root->left);
    postOrder(root->right);
    printf("%c ", root->data);
}

// 后序遍历（非递归）
void postOrderNonRecursive(TreeNode *root) {
    if (root == NULL) {
        return;
    }
    TreeNode *stack[100]; // 栈用于存储节点
    int top = -1;        // 栈顶指针
    TreeNode *lastVisited = NULL; // 记录上次访问的节点

    while (root != NULL || top != -1) {
        // 将左子树压入栈中
        while (root != NULL) {
            stack[++top] = root;
            root = root->left;
        }

        // 检查栈顶节点的右子树
        TreeNode *peekNode = stack[top];
        if (peekNode->right == NULL || peekNode->right == lastVisited) {
            // 如果右子树为空或者已经被访问过，则访问该节点
            printf("%c ", peekNode->data);
            lastVisited = stack[top--];
        } else {
            // 否则，处理右子树
            root = peekNode->right;
        }
    }
}

int main() {
    printf("请输入二叉树的节点（按层输入，空节点用 '#' 表示）:\n");
    TreeNode *root = createTree();

    printf("\n前序遍历（递归）: ");
    preOrder(root);

    printf("\n中序遍历（递归）: ");
    inOrder(root);

    printf("\n后序遍历（递归）: ");
    postOrder(root);

    printf("\n后序遍历（非递归）: ");
    postOrderNonRecursive(root);

    printf("\n");

    // 释放内存（可选，此处省略释放代码）
    return 0;
}
```

### 11.4 哈夫曼树

#### 11.4.1 基本概念

**路径：**从树中一个结点到另一个结点之间的分支构成这两个结点之间的路径

**路径长度：**路径上的分支数目

**树的路径长度：**从树根到每一个结点的路径长度之和

**结点的权：**在实际应用中，给树中的结点赋予代表某种含义的数值

**结点的带权路径长度：**从该结点到树根之间的路径长度与该结点权的乘积

**树的带权路径长度（WPL）：**书中所有叶节点的带权路径长度之和

---

带权路径长度**WPL**最小的二叉树称为哈夫曼树

#### 11.4.2 构造哈夫曼树

1.先把有权值的叶子结点按照从小到大的顺序排列成一个有序序列

2.取两个最小权值的结点作为一个新结点N1的子结点

3.将结点N1替换刚刚取出来的两个结点，并加入到有序序列中

4.重复以上步骤

## 12 串

### 12.1 串的定义

串：零个或多个字符组成的有限序列

串长度：串中所包含的字符个数

空串：长度为0的串，记为“”

非串通常记为：S=“s1s2.......sn”

S是串名，“”是定界符，双引号引起来的部分是串的值，si（1<=i<=n) 是任意一个字符

字符集：ASCII码，Unincode字符集等

子序列：是主串中的字符集合，这些字符的相对顺序与在主串中相同

字串：主串任意个连续字符组成的子序列

主串：包含子串的串

子串位置：子串第一个字符在主串中的位置序号

### 12.2 串的模式匹配算法

**什么是串的模式匹配**

子串定位运算又称为模式匹配或串匹配，此运算的应用非常广泛。例如，在文本编辑程序中，我们经常要查找某一特定单词在文本中出现的位置。显然，解词问题的有效算法能极大地提高文本编辑程序的响应性能

---

#### 12.2.1 朴素模式匹配算法（Brutr-Force算法）：枚举法

也就是俗称的**暴力算法**

在串匹配中一般将主串称为目标串，子串称之为模式串。设S为目标串，T为模式串

从主串S的第一个字符开始和模式T的第一个字符进行比较，若相等，则继续比较两者的后续字符；否则，从主串S的第二个字符开始和模式T的第一个字符进行比较，重复上述过程，直到T中的字符全部比较完毕，则说明本趟匹配成功；或S中的字符全部比较完，则说明匹配失败。

![](https://cdn.nlark.com/yuque/0/2025/png/53984905/1744160735217-652ddd22-4187-4973-808d-d7069590203f.png)

```c
#include <stdio.h>
#include <string.h>

/**
 * BF算法实现字符串模式匹配
 * @param S 主串
 * @param T 模式串
 * @return 匹配成功返回首次出现的位置（从0开始），失败返回-1
 */
int BF(char *S, char *T) {
    int lenS = strlen(S);        // 主串长度
    int lenT = strlen(T);        // 模式串长度
    
    // 处理特殊情况
    if (lenT == 0) return 0;     // 空模式串默认匹配位置0
    if (lenT > lenS) return -1;  // 模式串比主串长直接返回-1

    // 主串遍历起点i，最多只需到lenS-lenT位置
    for (int i = 0; i <= lenS - lenT; i++) {
        int j;
        // 检查模式串所有字符
        for (j = 0; j < lenT; j++) {
            if (S[i + j] != T[j]) {
                break;  // 字符不匹配时跳出
            }
        }
        // 完全匹配时返回当前起始位置
        if (j == lenT) {
            return i;
        }
    }
    return -1;  // 遍历完未找到匹配
}

int main() {
    char mainStr[] = "ababcabcacbab";  // 主串
    char pattern[] = "abcac";         // 模式串
    
    int position = BF(mainStr, pattern);
    
    if (position != -1) {
        printf("模式串首次出现在主串的位置：%d\n", position);
    } else {
        printf("模式串不在主串中\n");
    }
    return 0;
}
```

- **函数实现**：
- **参数**：主串S和模式串T
- **特殊处理**：

- 空模式串直接返回0
- 模式串比主串长直接返回-1

- **双重循环**：

- 外层循环控制主串起始位置（i）
- 内层循环进行逐字符比较（j）
- 发现不匹配立即跳出内层循环

#### 12.2.2 KMP算法

- **为什么BF算法的时间性能低？**

在每趟匹配不成功时存在大量**回溯**，没有利用**已经匹配的结果**

- **如何在匹配不成功时主串不回溯？**

主串不回溯，模式就需要滑动一段距离

- **如何确定模式的滑动距离**

利用已经得到的"部分匹配"的结果

将模式向右滑动尽可能远的一段距离（next[j]）后，继续进行比较

**出发点：利用前面匹配的结果，进行无回溯匹配**

**简单总结就是，BF算法每次往后移动一位，而KMP算法一次移动好几位**

---

以下是C语言实现KMP（Knuth-Morris-Pratt）字符串模式匹配算法的代码及解析：

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAXSIZE 100
//生成next数组,传入模式串
int *next_string(char *s){
	
	int length=strlen(s);//length为模式串长度
	
	int *next=(int *)malloc(sizeof(int)*length);
	next[0]=0;	
	
	int i=1;//一直往后移动至模式串末尾
	int j=0;//前缀指针
	
	while(i!=length){
		if(s[j]==s[i]){
			j++;
			next[i]=j;
		}
		else{
			j=next[j-1];
			next[i]=j;
		}
		i++;
	}
	
	return next;
}
//KMP匹配
int KMP(char *S,char *T){
	
	int *next_T=next_string(T);
	
	int length_S=strlen(S);
	int length_T=strlen(T);
	
	int i=0;//主串的指针，不会回退
	int j=0;//模式串的指针
	
	while(i < length_S && j < length_T){
		if(S[i]==T[j]){
			i++;
			j++;
		}
		else{
			
			if(j) j=next_T[j-1];
			else i++;
		}
	
	}
	
if (j == length_T) {
        return i - j;
    } else {
        return -1;
    }
}

int main(){
	char s[MAXSIZE];
	char t[MAXSIZE];
	
	printf("输入主串");
	scanf("%s",s);
	
	printf("输入模式串");
	scanf("%s",t);
	
	int is=KMP(s,t);
	
	if(is<0) printf("没有匹配到当前模式串");
	if(is>=0) printf("匹配成功，模式串在主串中的位置为%d",is);
	
	return 0;

}
```

### **代码解析**

#### **1. 核心思想**

- **避免无效匹配****：通过预处理生成**`**next**`**数组（部分匹配表），记录模式串的局部匹配信息**
- **智能跳转****：当发生失配时，根据**`**next**`**数组跳过不可能匹配的位置**

#### **2. next数组生成**

```c
void getNext(char *pattern, int *next) {
    int len = strlen(pattern);
    next[0] = -1;  // 初始化
    
    int i = 0;      // 后缀指针
    int j = -1;     // 前缀指针
    
    while (i < len - 1) {
        if (j == -1 || pattern[i] == pattern[j]) {
            i++;
            j++;
            // 优化：避免重复比较相同字符
            if (pattern[i] != pattern[j]) {
                next[i] = j;
            } else {
                next[i] = next[j];
            }
        } else {
            j = next[j];  // 关键回溯
        }
    }
}
```

- **计算原理****：**

- `**next[i]**`**表示当**`**pattern[i]**`**失配时，模式串应该跳转的位置**
- **通过比较前缀和后缀计算最长公共前后缀长度**

- **优化点****：**

- **当**`**pattern[i] == pattern[next[i]]**`**时，直接继承之前的跳转位置**

#### **3. 匹配过程**

```c
int KMP(char *text, char *pattern) {
    // ...初始化...
    while (i < textLen && j < patternLen) {
        if (j == -1 || text[i] == pattern[j]) {
            i++;
            j++;
        } else {
            j = next[j];  // 关键跳转
        }
    }
    // ...返回结果...
}
```

- **时间复杂度****：O(n+m)（n=主串长度，m=模式串长度）**
- **优势****：主串指针**`**i**`**永不回溯**


### **与BF算法对比**

|   |   |   |
|---|---|---|
|**特性**|**BF算法**|**KMP算法**|
|**时间复杂度**|**O(n*m)**|**O(n+m)**|
|**空间复杂度**|**O(1)**|**O(m)**|
|**主串指针回溯**|**需要**|**永不回溯**|
|**预处理**|**无**|**需要计算next数组**|
|**适用场景**|**短模式串**|**长模式串/重复多**|

### **关键图解说明（以模式串"abcac"为例）**

```
模式串: a b c a c
next数组: [-1,0,0,0,1]

生成过程：
1. next[0] = -1（初始化）
2. i=0, j=-1 → j=0, next[1]=0
3. i=1, j=0 → 'b'≠'a' → j=next[0]=-1
4. i=1, j=-1 → j=1, next[2]=0
5. i=2, j=0 → 'c'≠'a' → j=next[0]=-1
6. i=2, j=-1 → j=2, next[3]=0
7. i=3, j=0 → 'a'=='a' → next[4]=1
```

**该实现完整展示了KMP算法的核心思想，通过预处理模式串生成next数组，在匹配失败时利用已匹配信息实现智能跳转，显著提高了匹配效率。**
